cmake_minimum_required(VERSION 3.24)
project(deadzone_regressions LANGUAGES CXX)
enable_testing()
add_executable(deadzone_regressions test_native.cpp)
target_compile_features(deadzone_regressions PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(deadzone_regressions PRIVATE /W4 /UNDEBUG)
else()
  target_compile_options(deadzone_regressions PRIVATE -Wall -Wextra -UNDEBUG)
endif()
